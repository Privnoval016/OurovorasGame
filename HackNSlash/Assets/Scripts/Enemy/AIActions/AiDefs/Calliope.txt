// Calliope.aidef  --  Prologue Miniboss: Sweeping Striker
//
// ─── INSPECTOR SETTINGS ──────────────────────────────────────────────────────
//   EnemyStateMachine:
//     actionCooldownWindow = 1.5    (norm 1.0 = 1.5 s)
//     streakCap            = 3      (streak2 / cap3 = 0.67 -- kills at > 0.60)
//     meleeBurstCap        = 3      (burst2 / cap3 = 0.67 -- BackOff gate)
//   EnemySensor:
//     detectionRadius = 10          (norm 1.0 = 10 m)
//
// ─── DISTANCE REFERENCE ──────────────────────────────────────────────────────
//   0.12 norm =  1.2 m  lunge arrival / tight melee
//   0.35 norm =  3.5 m  BackOff distance gate
//   0.40 norm =  4.0 m  melee attack outer boundary
//   0.55 norm =  5.5 m  lunge sweet spot
//   0.65 norm =  6.5 m  orbit centre
//   1.00 norm = 10.0 m  sensor edge
//
// ─── PHASE FLOW (repeating) ──────────────────────────────────────────────────
//   1. OrbitFar at 6.5 m (score 0.65) for 2 s
//   2. ForwardLunge once timer > 0.75 (score 0.72 beats orbit 0.65)
//   3. CloseDash fills until timer > 0.40 (score 0.30, no burst reset)
//   4. Sweep (burst → 2, score ~0.80)
//   5. PostAttackPause 0.3 s (score 0.55 beats CloseDash 0.30)
//   6. Overhead (burst → 3, score ~0.94 with StringMatch)
//   7. PostAttackPause 0.3 s
//   8. BackOff: burst ≥ 2 → norm ≥ 0.67 > 0.60 → score 1.5 HARD BEATS all attacks
//      (fires after Sweep when burst=2 if attacks are blocked, guaranteed after Overhead)
//   9. BackOff retreats to 6.5 m, resets burst to 0 → OrbitFar picks up → cycle repeats
//
// ─── WHY BackOff uses consecutiveMeleeCountNorm, not timer ───────────────────
//   The timer resets on every attack START, so after Sweep it reads 0, then 0.40
//   by the time the next action evaluates -- never reaching a "high" threshold
//   without blocking all attacks. ConsecutiveMeleeCountNorm counts total attack
//   commits regardless of which attack, giving a clean "after N hits" gate that
//   is independent of animation length and timer drift.

// ════════════════════════════════════════════════════════════════
// 1. Shield-Break Stun
// ════════════════════════════════════════════════════════════════
action "CalliopeShieldBreakStunned" : EnemyStunned {
  exitCondition = ShieldRestored

  consideration composite {
    allMustBeNonZero = true
    first bool { key = "bool.is_shielded" invert = true }
    rest [
      multiply: constant { value = 10 }
    ]
  }
}

// ════════════════════════════════════════════════════════════════
// 2. Post-Attack Pause  (0.3 s breather after each melee hit)
//
//    Triggered by StringMatch on last action name -- reliable because
//    the timer resets on attack START not END, making timer-based
//    triggers miss the window. StringMatch fires exactly when a melee
//    attack just completed regardless of animation length.
//
//    Score 0.55 > CloseDash 0.30 at dist ≤ 0.45.
//    ResetsActionTimer = false → _lastActionName is preserved through
//    the pause so the next brain tick still sees the correct history
//    for the Sweep ↔ Overhead StringMatch alternation.
// ════════════════════════════════════════════════════════════════
action "CalliopePostAttackPause" : EnemyIdle {
  duration   = 0.3
  facePlayer = true

  consideration composite {
    allMustBeNonZero = true
    first bool { key = "bool.is_shielded" }
    rest [
      multiply: bool { key = "bool.is_staggered" invert = true },
      multiply: curve {
        key    = "float.dist_to_player_norm"
        points = [[0.0, 1.0], [0.45, 1.0], [0.50, 0.0], [1.0, 0.0]]
      },
      multiply: composite {
        allMustBeNonZero = false
        first stringmatch {
          key       = "history.last_action_name"
          match     = "CalliopeSweepAttack"
          ifmatch   = 0.55
          ifnomatch = 0.0
        }
        rest [
          add: stringmatch {
            key       = "history.last_action_name"
            match     = "CalliopeOverheadStrike"
            ifmatch   = 0.55
            ifnomatch = 0.0
          }
        ]
      }
    ]
  }
}

// ════════════════════════════════════════════════════════════════
// 3. Back Off  (hard retreat after melee burst ≥ 2)
//
//    Uses consecutiveMeleeCountNorm > 0.60 (= burst ≥ 2 with cap=3).
//    Score = 1.5 -- intentionally beats ALL attacks (max ~0.94) and
//    PostAttackPause (0.55) so the retreat is guaranteed once the
//    burst threshold is reached.
//
//    Only fires at dist ≤ 0.35 (3.5 m) -- she must be in melee range.
//    At orbit range dist > 0.50 the curve returns 0 so BackOff
//    cannot interrupt orbiting or strafing.
//
//    Streak gate blocks back-to-back BackOff.
//    After BackOff, movement action resets _consecutiveMeleeCount to 0
//    via the BreaksMeleeBurst flag so the burst counter clears cleanly.
//
//    Inspector: set BreaksMeleeBurst = true on this action asset.
// ════════════════════════════════════════════════════════════════
action "CalliopeBackOff" : EnemyMovement {
  strategy         = Retreat
  speed            = 1.2
  desiredDist      = 6.5
  maxDuration      = 2.0
  breaksMeleeBurst = true

  consideration composite {
    allMustBeNonZero = true
    first bool { key = "bool.is_shielded" }
    rest [
      multiply: bool         { key = "bool.is_staggered"  invert = true },
      multiply: targetexists { key = "target.player" },
      multiply: curve {
        key    = "float.dist_to_player_norm"
        points = [[0.0, 1.0], [0.35, 1.0], [0.45, 0.0], [1.0, 0.0]]
      },
      multiply: threshold {
        key        = "float.consecutive_melee_count_norm"
        comparison = greaterthan
        threshold  = 0.60
        iftrue     = 1.5
        iffalse    = 0.0
      },
      multiply: threshold {
        key        = "float.same_action_streak_norm"
        comparison = greaterthan
        threshold  = 0.60
        iftrue     = 0.0
        iffalse    = 1.0
      }
    ]
  }
}
